I = range(1, 64)
J = range(1, 37)
W = range(1, 37)

t_raw = {
    1: 2.43,  2: 9.79,  3: 2.12,  4: 9.92,  5: 4.66,  6: 11.58,
    7: 1.01,  8: 1.44,  9: 9.66, 10: 10.30, 11: 0.49, 12: 7.13,
    13: 7.18, 14: 2.44, 15: 3.58, 16: 4.90, 17: 3.21, 18: 7.78,
    19: 11.27, 20: 11.35, 21: 0.80, 22: 3.31, 23: 9.83, 24: 0.80,
    25: 4.61, 26: 5.20, 27: 11.89, 28: 6.30, 29: 13.32, 30: 0.98,
    31: 14.20, 32: 6.13, 33: 0.98, 34: 14.49, 35: 3.14, 36: 12.12,
    37: 1.07, 38: 5.14, 39: 5.63, 40: 0.57, 41: 10.13, 42: 0.90,
    43: 1.39, 44: 1.43, 45: 0.51, 46: 10.74, 47: 5.65, 48: 7.38,
    49: 1.71, 50: 15.09, 51: 7.31, 52: 6.93, 53: 10.72, 54: 1.31,
    55: 6.45, 56: 2.39, 57: 0.89, 58: 11.06, 59: 8.02, 60: 6.48,
    61: 3.13, 62: 0.53, 63: 7.74
}

SCALE = 100
t = {i: int(round(t_raw[i] * SCALE)) for i in t_raw}
P = [(i, i + 1) for i in range(1, 63)]
d = {j: {k: 2 * abs(j - k) for k in J} for j in J}
BIG_M = sum(t.values())

# =========================================================
# MODEL KURAN FONKSIYON
# =========================================================
def solve_model(exact_workers=None, time_limit=5, upper_bound_C=None):
    model = cp_model.CpModel()

    x = {(i, j): model.NewBoolVar(f"x_{i}_{j}") for i in I for j in J}
    y = {(w, j): model.NewBoolVar(f"y_{w}_{j}") for w in W for j in J}
    z = {w: model.NewBoolVar(f"z_{w}") for w in W}

    l = {j: model.NewIntVar(0, BIG_M, f"l_{j}") for j in J}
    q = {(w, j): model.NewIntVar(0, BIG_M, f"q_{w}_{j}") for w in W for j in J}
    C = model.NewIntVar(0, BIG_M, "C")

    if upper_bound_C is not None:
        model.Add(C <= int(round(upper_bound_C * SCALE)))

    for i in I:
        model.Add(sum(x[i, j] for j in J) == 1)

    for i, h in P:
        model.Add(sum(j * x[i, j] for j in J) <= sum(j * x[h, j] for j in J))

    for j in J:
        model.Add(l[j] == sum(t[i] * x[i, j] for i in I))

    for j in J:
        model.Add(sum(y[w, j] for w in W) == 1)

    for w in W:
        for j in J:
            model.Add(y[w, j] <= z[w])

    for w in W:
        for j in J:
            model.Add(q[w, j] <= l[j])
            model.Add(q[w, j] <= BIG_M * y[w, j])
            model.Add(q[w, j] >= l[j] - BIG_M * (1 - y[w, j]))

    for w in W:
        model.Add(sum(q[w, j] for j in J) <= C)

    for j in J:
        model.Add(l[j] <= C)

    for w in W:
        for j in J:
            for k in J:
                if j < k and d[j][k] > L:
                    model.Add(y[w, j] + y[w, k] <= 1)

    if exact_workers is not None:
        model.Add(sum(z[w] for w in W) == exact_workers)

    model.Minimize(C)

    solver = cp_model.CpSolver()
    solver.parameters.max_time_in_seconds = time_limit
    solver.parameters.num_search_workers = 8

    status = solver.Solve(model)

    if status not in [cp_model.OPTIMAL, cp_model.FEASIBLE]:
        return None

    C_value = solver.Value(C) / SCALE

    solution = {
        "C": C_value,
        "used_workers": sum(solver.Value(z[w]) for w in W),
        "stations_of_worker": {w: [] for w in W},
        "ops_of_station": {j: [] for j in J},
        "station_loads": {j: solver.Value(l[j]) / SCALE for j in J},
        "worker_load_per_product": {w: sum(solver.Value(q[w, j]) for j in J) / SCALE for w in W},
        "worker_load_per_shift": {w: D * sum(solver.Value(q[w, j]) for j in J) / SCALE for w in W},
        "worker_U": {w: 100 * ((D / T) * (sum(solver.Value(q[w, j]) for j in J) / SCALE)) for w in W},
        "reachable_output": T / C_value if C_value > 0 else float("inf"),
        "meets_target": (T / C_value >= D - 1e-6) if C_value > 0 else True,
    }

    for i in I:
        for j in J:
            if solver.Value(x[i, j]) == 1:
                solution["ops_of_station"][j].append(i)

    for w in W:
        for j in J:
            if solver.Value(y[w, j]) == 1:
                solution["stations_of_worker"][w].append(j)

    return solution

# ANA BUTON VE HESAPLAMA SÜRECİ
# =========================================================
if st.button("🚀 Tüm Senaryoları Hesapla ve Analiz Et"):
    
    results = {}
    best_C_so_far = None

    # Optimizasyon durumu için bir yükleme çubuğu ekleyelim
    progress_bar = st.progress(0)
    status_text = st.empty()

    # 1'den 11'e kadar olan senaryolar otomatik infeasible
    for eps in range(1, 12):
        results[eps] = None

    # 12'den 36'ya kadar olan senaryoları çöz
    total_scenarios = 36 - 12 + 1
    for idx, eps in enumerate(range(12, 37)):
        status_text.text(f"⏳ {eps} operatörlü senaryo çözülüyor...")
        
        # Streamlit arayüzünün akıcı olması için her senaryoya maks 3 saniye süre verdik
        res = solve_model(
            exact_workers=eps,
            time_limit=3, 
            upper_bound_C=best_C_so_far
        )
        
        if res is not None:
            best_C_so_far = res["C"]
            
        results[eps] = res
        progress_bar.progress(int((idx + 1) / total_scenarios * 100))

    status_text.success("✅ Tüm senaryolar başarıyla hesaplandı!")
    progress_bar.empty()

    # Model sonuçlarından İdeal ve Nadir Noktaları bulalım
    feasible_eps = [e for e, res in results.items() if res is not None]
    
    if feasible_eps:
        ideal_C = min(results[e]["C"] for e in feasible_eps)
        ideal_Z = min(results[e]["used_workers"] for e in feasible_eps)
        nadir_C = max(results[e]["C"] for e in feasible_eps)
        nadir_Z = max(results[e]["used_workers"] for e in feasible_eps)
        
        # =========================================================
        # GÖRSEL 1: ÖZET TABLO VE NOKTALAR
        # =========================================================
        st.header("📋 ÖZET TABLO")
        st.markdown(f"**İdeal Nokta** = ({ideal_C:.2f}, {ideal_Z:.2f})")
        st.markdown(f"**Nadir Nokta** = ({nadir_C:.2f}, {nadir_Z:.2f})")
        
        # Terminal çıktısı formatında tablo tasarımı (Birebir görseldeki gibi)
        summary_lines = []
        summary_lines.append(f"{'Epsilon':<10} | {'F1 (C)':<10} | {'F2 (Z)':<10} | {'Ulaşılabilir Üretim':<20} | {'Hedef?':<8}")
        summary_lines.append("-" * 75)
        
        # 1-11 Arası Infeasible Satırları
        for eps in range(1, 12):
            summary_lines.append(f"{eps:<10.2f} | {'Infeasible':<10} | {'-':<10} | {'-':<20} | {'-':<8}")
            
        # 12-36 Arası Gerçek Sonuçlar
        for eps in sorted(results.keys()):
            res = results[eps]
            if res is not None:
                cap = res["reachable_output"]
                meets = "Evet" if res["meets_target"] else "Hayır"
                summary_lines.append(
                    f"{eps:<10.2f} | {res['C']:<10.2f} | {res['used_workers']:<10.2f} | {cap:<20.2f} | {meets:<8}"
                )
        
        st.code("\n".join(summary_lines), language="text")

        # =========================================================
        # GÖRSEL 2-3-4-5: DETAYLI SENARYO RAPORU
        # =========================================================
        st.write("---")
        st.header(f"📊 DETAYLI SENARYO RAPORU | Operatör Sayısı = {epsilon_choice}")
        
        chosen_res = results.get(epsilon_choice)
        
        if chosen_res is None:
            st.error(f"Seçilen {epsilon_choice} operatör sayısı için uygun bir çözüm bulunamadı (Infeasible).")
        else:
            # Üst Özet Bilgiler (Görsel 2'deki gibi)
            report_header = [
                f"{'Çevrim Süresi (C)':<43}: {chosen_res['C']:.2f} dk/ürün",
                f"{'Kullanılan Operatör Sayısı':<43}: {chosen_res['used_workers']}",
                f"{'Maksimum İzin Verilen Operatör Doluluğu':<43}: %{U_MAX*100:.2f}",
                f"{'Ulaşılabilir Üretim':<43}: {chosen_res['reachable_output']:.2f} adet/vardiya",
                f"{'Hedef Üretim ({D} adet) Sağlanıyor mu?':<43}: {'Evet' if chosen_res['meets_target'] else 'Hayır'}"
            ]
            st.code("\n".join(report_header), language="text")
            
            # Yan yana kolonlar halinde detay çıktıları basıyoruz
            col1, col2 = st.columns(2)
            col3, col4 = st.columns(2)
            
            # [1] Operasyon -> İstasyon Atamaları (Görsel 2)
            with col1:
                st.subheader("[1] Operasyon -> İstasyon Atamaları")
                op_lines = []
                for j in J:
                    ops = chosen_res["ops_of_station"][j]
                    if ops:  # Eğer istasyona atanan iş varsa listele
                        op_lines.append(f"İstasyon {j}: {ops}")
                st.code("\n".join(op_lines), language="text")
                
            # [2] İstasyon Yükleri (Görsel 3)
            with col2:
                st.subheader("[2] İstasyon Yükleri")
                load_lines = []
                for j in J:
                    if chosen_res["ops_of_station"][j]:
                        load_lines.append(f"İstasyon {j}: {chosen_res['station_loads'][j]:.2f} dk")
                st.code("\n".join(load_lines), language="text")
                
            # [3] Operatör -> İstasyon Atamaları (Görsel 4)
            with col3:
                st.subheader("[3] Operatör -> İstasyon Atamaları")
                worker_lines = []
                for w in W:
                    stat = chosen_res["stations_of_worker"][w]
                    if stat:
                        worker_lines.append(f"Operatör {w}: {stat}")
                st.code("\n".join(worker_lines), language="text")
                
            # [4] Operatör Toplam Yükleri ve U Değerleri (Görsel 5)
            with col4:
                st.subheader("[4] Operatör Toplam Yükleri ve U Değerleri")
                u_lines = []
                for w in W:
                    if chosen_res["stations_of_worker"][w]:
                        per_prod = chosen_res['worker_load_per_product'][w]
                        per_shift = chosen_res['worker_load_per_shift'][w]
                        u_val = chosen_res['worker_U'][w]
                        u_lines.append(f"Operatör {w}: ürün başı yük = {per_prod:.2f} dk, vardiya yükü = {per_shift:.2f} dk, U = %{u_val:.2f}")
                st.code("\n".join(u_lines), language="text")
    else:
